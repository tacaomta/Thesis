\section{Algorithm for Synthetic Dataset Generation Based on the Barabási–Albert Model}

\subsection*{Algorithm S1}
\begin{algorithm}
\caption{RandomUpdateFunction}
\label{alg:random_update_function}
\begin{algorithmic}[1]
\Require Graph $G(V, A)$
\For{\textbf{each} $v \in V$}
    \State let $u_1, u_2, \dots, u_k$ genes regulating $v$ in $G$
    \State let $l_1, l_2, \dots, l_k, l$ the discretization level of $u_1, u_2, \dots, u_k$  and $v$
    \For{\textbf{each} $x \in \{0, \dots, l_1 - 1\} \times \{0, \dots, l_2 - 1\} \times \dots \times \{0, \dots, l_k - 1\}$}
        \State $F_v(x) \gets \text{randint}(0, l - 1)$ \Comment{a random integer between 0 and $l-1$}
    \EndFor
\EndFor
\State \Return $\{F_v | v \in V\}$
\end{algorithmic}
\end{algorithm}
\par

\subsection*{Algorithm S2}
\begin{algorithm}
\caption{RandomGeneExpression}
\label{alg:random_gene_exression}
\begin{algorithmic}[1]
\Require Graph $G(V, A)$, update functions $F_v$, observed time step $K$
\For{\textbf{each} $v \in V$}
    \State let the discretization level of $v$
    \State $v(0) \gets \text{randint}(0, l - 1)$\Comment{a random integer between 0 and $l-1$}
\EndFor
\For{$t = 1$ to $K$}
    \For{each $v \in V$}
        \State let $v$: $u_1, u_2, \dots, u_k$ genes regulating $v$ in $G$
        \State $x = (u_1(t-1), u_2(t-1), \dots, u_k(t-1))$
        \State $v(t+1) \gets F_v(x)$
    \EndFor
\EndFor
\State \Return $\{v(t) | v \in V, t \in \{0, \dots, K\}\}$
\end{algorithmic}
\end{algorithm}
